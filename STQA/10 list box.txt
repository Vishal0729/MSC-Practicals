import io.github.bonigarcia.wdm.WebDriverManager;
import org.openqa.selenium.By;
import org.openqa.selenium.WebDriver;
import org.openqa.selenium.WebElement;
import org.openqa.selenium.chrome.ChromeDriver;
import org.openqa.selenium.support.ui.Select;

import java.util.List;

public class Ver {

    public static void main(String[] args) {

        // Setup ChromeDriver
        WebDriverManager.chromedriver().setup();

        // Launch Chrome Browser
        WebDriver driver = new ChromeDriver();
        driver.manage().window().maximize();

        // Open the website
        driver.get("https://demoqa.com/select-menu");

        // Locate the dropdown
        WebElement dropdown = driver.findElement(By.id("oldSelectMenu"));

        // Create Select object
        Select select = new Select(dropdown);

        // Get all options
        List<WebElement> options = select.getOptions();

        // Display total number of items
        System.out.println("Total number of items: " + options.size());

        // Display all items
        System.out.println("\nDropdown Items:");
        for (WebElement option : options) {
            System.out.println(option.getText());
        }

        // Display default selected value
        String defaultValue = select.getFirstSelectedOption().getText();
        System.out.println("\nDefault Selected Value: " + defaultValue);

        // Select by Visible Text
        select.selectByVisibleText("Blue");
        System.out.println("After selectByVisibleText(): "
                + select.getFirstSelectedOption().getText());

        // Select by Value
        select.selectByValue("5");
        System.out.println("After selectByValue(): "
                + select.getFirstSelectedOption().getText());

        // Select by Index
        select.selectByIndex(2);
        System.out.println("After selectByIndex(): "
                + select.getFirstSelectedOption().getText());

        // Close browser
        driver.quit();
    }
}
