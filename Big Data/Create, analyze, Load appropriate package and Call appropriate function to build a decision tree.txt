
# Decision Tree Classification

## Python

```python
# ===== Step 1: Load appropriate packages (5 Marks) =====
import pandas as pd
from sklearn.tree import DecisionTreeClassifier, plot_tree, export_text
from sklearn.model_selection import train_test_split
from sklearn.metrics import accuracy_score, classification_report
import matplotlib.pyplot as plt

# ===== Step 2: Load Dataset =====
from sklearn.datasets import load_iris
iris = load_iris()
X = pd.DataFrame(iris.data, columns=iris.feature_names)
y = iris.target

print("Dataset: Iris")
print(f"Samples: {len(X)}, Features: {X.shape[1]}")
print(f"Classes: {list(iris.target_names)}")

# ===== Step 3: Split Data =====
X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.3, random_state=42)
print(f"\nTrain: {len(X_train)}, Test: {len(X_test)}")

# ===== Step 4: Build Decision Tree (5 Marks) =====
dt = DecisionTreeClassifier(criterion='gini', max_depth=3, random_state=42)
dt.fit(X_train, y_train)
print("\nDecision Tree built successfully!")

# ===== Step 5: Analyze the Tree (5 Marks) =====
# Text representation
print("\nTree Structure:")
print(export_text(dt, feature_names=iris.feature_names))

# Predictions & Accuracy
y_pred = dt.predict(X_test)
print(f"Accuracy: {accuracy_score(y_test, y_pred) * 100:.2f}%")
print("\nClassification Report:")
print(classification_report(y_test, y_pred, target_names=iris.target_names))

# Feature Importance
print("Feature Importance:")
for name, imp in zip(iris.feature_names, dt.feature_importances_):
    print(f"  {name}: {imp:.4f}")

# ===== Step 6: Plot the Tree =====
plt.figure(figsize=(12, 7))
plot_tree(dt, feature_names=iris.feature_names,
          class_names=list(iris.target_names), filled=True, rounded=True)
plt.title("Decision Tree - Iris Dataset")
plt.tight_layout()
plt.savefig("decision_tree.png")
plt.show()
```

## R

```r
# ===== Step 1: Load appropriate packages (5 Marks) =====
library(rpart)         # Decision tree
library(rpart.plot)    # Tree plotting

# ===== Step 2: Load Dataset =====
data(iris)
cat("Dataset: Iris\n")
cat(sprintf("Samples: %d, Features: %d\n", nrow(iris), ncol(iris) - 1))
cat("Classes:", levels(iris$Species), "\n")

# ===== Step 3: Split Data =====
set.seed(42)
train_idx <- sample(1:nrow(iris), 0.7 * nrow(iris))
train_data <- iris[train_idx, ]
test_data <- iris[-train_idx, ]
cat(sprintf("\nTrain: %d, Test: %d\n", nrow(train_data), nrow(test_data)))

# ===== Step 4: Build Decision Tree (5 Marks) =====
dt <- rpart(Species ~ ., data = train_data, method = "class")
cat("\nDecision Tree built successfully!\n")

# ===== Step 5: Analyze the Tree (5 Marks) =====
# Text representation
cat("\nTree Structure:\n")
print(dt)

# Predictions & Accuracy
y_pred <- predict(dt, test_data, type = "class")
accuracy <- sum(y_pred == test_data$Species) / nrow(test_data)
cat(sprintf("\nAccuracy: %.2f%%\n", accuracy * 100))

# Confusion Matrix
cat("\nConfusion Matrix:\n")
print(table(Predicted = y_pred, Actual = test_data$Species))

# Feature Importance
cat("\nFeature Importance:\n")
print(dt$variable.importance)

# ===== Step 6: Plot the Tree =====
rpart.plot(dt, main = "Decision Tree - Iris Dataset",
           extra = 104, under = TRUE)
```
